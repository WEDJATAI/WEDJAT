   110→function LifecycleStepper({ status }: { status: string }) {
   111→  const cur = LIFECYCLE.indexOf(status as (typeof LIFECYCLE)[number]);
   112→  const terminal = cur === -1;
   113→  return (
   114→    <div
   115→      className="wedjat-scroll flex items-center gap-1 overflow-x-auto py-1"
   116→      aria-label={`Run lifecycle, current status ${status}`}
   117→    >
   118→      {LIFECYCLE.map((step, i) => {
   119→        const state = terminal
   120→          ? "pending"
   121→          : i < cur
   122→            ? "done"
   123→            : i === cur
   124→              ? "current"
   125→              : "pending";
   126→        return (
   127→          <div key={step} className="flex shrink-0 items-center gap-1">
   128→            {i > 0 ? (
   129→              <span
   130→                aria-hidden="true"
   131→                className={cn(
   132→                  "h-px w-5",
   133→                  state === "pending" ? "bg-border" : "bg-primary/40",
   134→                )}
   135→              />
   136→            ) : null}
   137→            <span
   138→              className={cn(
   139→                "flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-[10px] font-medium",
   140→                state === "done" &&
   141→                  "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400",
   142→                state === "current" &&
   143→                  "wedjat-pulse border-amber-500/50 bg-amber-500/15 text-amber-700 dark:text-amber-400",
   144→                state === "pending" &&
   145→                  "border-border text-muted-foreground/70",
   146→              )}
   147→            >
   148→              {state === "done" ? (
   149→                <CheckCircle2 aria-hidden="true" className="size-3" />
   150→              ) : null}
   151→              {step}
   152→            </span>
   153→          </div>
   154→        );
   155→      })}
   156→      {terminal ? <StatusBadge status={status} /> : null}
   157→    </div>
   158→  );
   159→}

   442→function CreateDatasetForm({
   443→  eligibleKinds,
   444→  onCreated,
   445→}: {
   446→  eligibleKinds: string[];
   447→  onCreated: () => void;
   448→}) {
   449→  const [name, setName] = useState("");
   450→  const [description, setDescription] = useState("");
   451→  const [kinds, setKinds] = useState<string[]>([...TRAINING_KINDS]);
   452→  const [minQuality, setMinQuality] = useState(0);
   453→  const [submitting, setSubmitting] = useState(false);
   454→  const [error, setError] = useState<string | null>(null);
   455→
   456→  const toggleKind = (k: string, checked: boolean) => {
   457→    setKinds((cur) =>
   458→      checked ? [...new Set([...cur, k])] : cur.filter((x) => x !== k),
   459→    );
   460→  };
   461→
   462→  const submit = async () => {
   463→    if (!name.trim() || submitting) return;
   464→    setSubmitting(true);
   465→    setError(null);
   466→    try {
   467→      const res = await apiPost<{
   468→        datasetId: string;
   469→        datasetVersionId: string;
   470→        exampleCount: number;
   471→        excluded: number;
   472→        reasons: string[];
   473→      }>("/api/training", {
   474→        action: "create-dataset",
   475→        name: name.trim(),
   476→        description: description.trim() || undefined,
   477→        includeKinds: kinds,
   478→        minQuality,
   479→      });
   480→      toast.success("Dataset created", {
   481→        description: `${res.exampleCount} examples included, ${res.excluded} excluded${
   482→          res.reasons.length > 0 ? ` — ${res.reasons.join("; ")}` : ""
   483→        }.`,
   484→      });
   485→      setName("");
   486→      setDescription("");
   487→      onCreated();
   488→    } catch (e) {
   489→      setError(errMessage(e));
   490→    } finally {
   491→      setSubmitting(false);
   492→    }
   493→  };
   494→
   495→  return (
   496→    <Card className="rounded-xl">
   497→      <CardContent className="space-y-4 p-4 sm:p-6">
   498→        <div className="space-y-1.5">
   499→          <Label htmlFor="ds-name">Dataset name</Label>
   500→          <Input
   501→            id="ds-name"
   502→            value={name}
   503→            onChange={(e) => setName(e.target.value)}
   504→            placeholder="e.g. checkout-domain-v1"
   505→            className="h-9"
   506→          />
   507→        </div>
   508→        <div className="space-y-1.5">
   509→          <Label htmlFor="ds-desc">Description (optional)</Label>
   510→          <Textarea
   511→            id="ds-desc"
   512→            value={description}
   513→            onChange={(e) => setDescription(e.target.value)}
   514→            placeholder="What this dataset trains or evaluates…"
   515→            className="min-h-16 text-sm"
   516→          />
   517→        </div>
   518→        <fieldset className="space-y-2">
   519→          <legend className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
   520→            Include kinds
   521→          </legend>
   522→          <div className="flex flex-wrap gap-4">
   523→            {TRAINING_KINDS.map((k) => (
   524→              <label
   525→                key={k}
   526→                className="flex min-h-11 items-center gap-2 text-sm"
   527→              >
   528→                <Checkbox
   529→                  checked={kinds.includes(k)}
   530→                  onCheckedChange={(c) => toggleKind(k, c === true)}
   531→                  aria-label={`Include ${k} examples`}
   532→                />
   533→                {k}
   534→                <span className="font-mono text-[10px] text-muted-foreground">
   535→                  {eligibleKinds.includes(k) ? "" : "(none eligible yet)"}
   536→                </span>
   537→              </label>
   538→            ))}
   539→          </div>
   540→        </fieldset>
   541→        <div className="space-y-2">
   542→          <div className="flex items-center justify-between">
   543→            <Label htmlFor="ds-quality">Minimum quality score</Label>
   544→            <span className="font-mono text-xs tabular-nums text-muted-foreground">
   545→              {minQuality.toFixed(2)}
   546→            </span>
   547→          </div>
   548→          <Slider
   549→            id="ds-quality"
   550→            value={[minQuality]}
   551→            onValueChange={([v]) => setMinQuality(v)}
   552→            min={0}
   553→            max={1}
   554→            step={0.05}
   555→            aria-label="Minimum quality score"
   556→          />
   557→          <p className="text-xs text-muted-foreground">
   558→            Examples below this score are excluded (with reasons reported back).
   559→          </p>
   560→        </div>
   561→        {error ? (
   562→          <p className="text-sm text-red-700 dark:text-red-400">{error}</p>
   563→        ) : null}
   564→        <Button
   565→          onClick={submit}
   566→          disabled={!name.trim() || kinds.length === 0 || submitting}
   567→        >
   568→          {submitting ? (
   569→            <>
   570→              <LoaderCircle className="size-4 animate-spin" aria-hidden="true" />
   571→              Building…
   572→            </>
   573→          ) : (
   574→            <>
   575→              <Database className="size-4" aria-hidden="true" />
   576→              Create dataset
   577→            </>
   578→          )}
   579→        </Button>
   580→      </CardContent>
   581→    </Card>
   582→  );
   583→}

   585→function StartRunForm({
   586→  datasets,
   587→  onStarted,
   588→}: {
   589→  datasets: TrainingDatasetDto[];
   590→  onStarted: () => void;
   591→}) {
   592→  const options = datasets.filter((d) => d.currentVersion);
   593→  const [datasetVersionId, setDatasetVersionId] = useState("");
   594→  const [method, setMethod] = useState<string>("SIMULATED");
   595→  const [submitting, setSubmitting] = useState(false);
   596→  const [error, setError] = useState<string | null>(null);
   597→
   598→  const submit = async () => {
   599→    if (!datasetVersionId || submitting) return;
   600→    setSubmitting(true);
   601→    setError(null);
   602→    try {
   603→      const res = await apiPost<{ runId: string; jobId: string }>(
   604→        "/api/training",
   605→        { action: "start-run", datasetVersionId, method },
   606→      );
   607→      toast.success("Training run started", {
   608→        description: `Run ${res.runId.slice(0, 8)}… (job ${res.jobId.slice(0, 8)}…). The worker auto-progresses to EVALUATING; promotion gates stay manual.`,
   609→      });
   610→      onStarted();
   611→    } catch (e) {

   690→export function TrainingView({ role }: { role: string }) {
   691→  const [data, setData] = useState<TrainingPayload | null>(null);
   692→  const [error, setError] = useState<string | null>(null);
   693→  const [loading, setLoading] = useState(true);
   694→  const pollRef = useRef<() => void>(() => {});
   695→
   696→  const load = async () => {
   697→    try {
   698→      const d = await api<TrainingPayload>("/api/training");
   699→      setData(d);
   700→      setError(null);
   701→    } catch (e) {
   702→      setError(errMessage(e));
   703→    } finally {
   704→      setLoading(false);
   705→    }
   706→  };
   707→
   708→  useEffect(() => {
   709→    load();
   710→  }, []);
   711→
   712→  // Poll every 3s while any run is in an active lifecycle stage.
   713→  useEffect(() => {
   714→    let interval: ReturnType<typeof setInterval> | undefined;
   715→    let stopped = false;
   716→    const tick = async () => {
   717→      if (stopped) return;
   718→      try {
   719→        const d = await api<TrainingPayload>("/api/training");
   720→        if (stopped) return;
   721→        setData(d);
   722→        setError(null);
   723→        const active = d.runs.some((r) => ACTIVE_RUN_STATUSES.has(r.status));
   724→        if (active && !interval) {
   725→          interval = setInterval(tick, 3000);
   726→        } else if (!active && interval) {
   727→          clearInterval(interval);
   728→          interval = undefined;
   729→        }
   730→      } catch {
   731→        /* keep last state */
   732→      }
   733→    };
   734→    pollRef.current = tick;
   735→    // Initial check arms the interval if runs are already in flight
   736→    // (e.g. the user navigated away and back).
   737→    tick();
   738→    return () => {
   739→      stopped = true;
   740→      if (interval) clearInterval(interval);
   741→    };
   742→  }, []);
   743→
   744→  const canManage = canManageTraining(role);
   745→
   746→  if (loading && !data) {
   747→    return (
   748→      <div className="space-y-6">
   749→        <SectionHeading eyebrow="Lifecycle" title="Training" />
   750→        <SkeletonRows rows={5} />
   751→      </div>
   752→    );
   753→  }
   754→
   755→  if (error && !data) {
   756→    return (
   757→      <div className="space-y-6">
   758→        <SectionHeading eyebrow="Lifecycle" title="Training" />
   759→        <ErrorState message={error} onRetry={load} />
   760→      </div>
   761→    );
   762→  }
   763→
   764→  const d = data;
   765→  if (!d) return null;

   769→      <SectionHeading
   770→        eyebrow="Lifecycle"
   771→        title="Training"
   772→        description="Datasets from feedback and synthetic sources, quality-gated; runs advance to EVALUATING automatically — promotion to CANDIDATE/CANARY/PRODUCTION is always an explicit human action."
   773→        actions={
   774→          <Button variant="outline" size="sm" className="h-9" onClick={load}>
   775→            <Sparkles className="size-3.5" aria-hidden="true" />
   776→            Refresh
   777→          </Button>
   778→        }
   779→      />
   780→
   781→      <motion.div
   782→        initial={{ opacity: 0, y: 6 }}
   783→        animate={{ opacity: 1, y: 0 }}
   784→        role="status"
   785→        aria-label="Simulated lifecycle notice"
   786→        className="flex items-start gap-3 rounded-xl border border-amber-500/50 bg-amber-500/10 p-4"
   787→      >
   788→        <TriangleAlert
   789→          aria-hidden="true"
   790→          className="mt-0.5 size-5 shrink-0 text-amber-600 dark:text-amber-400"
   791→        />
   792→        <div>
   793→          <p className="text-sm font-semibold text-amber-800 dark:text-amber-400">
   794→            SIMULATED LIFECYCLE
   795→          </p>
   796→          <p className="mt-1 text-sm text-amber-800/80 dark:text-amber-400/80">

   800→            weights are ever trained or deployed. The UI labels this honestly
   801→            everywhere.
   802→          </p>

   806→      {error ? (

   811→        <section aria-label="Eligible sources" className="space-y-3">
