   130→/** Collapsible card section with eyebrow + title + description. */
   131→function DetailSection({
   132→  eyebrow,
   133→  title,
   134→  description,
   135→  defaultOpen = true,
   136→  open,
   137→  onOpenChange,
   138→  children,
   139→  id,
   140→}: {
   141→  eyebrow?: string;
   142→  title: string;
   143→  description?: string;
   144→  defaultOpen?: boolean;
   145→  open?: boolean;
   146→  onOpenChange?: (open: boolean) => void;
   147→  children: React.ReactNode;
   148→  id?: string;
   149→}) {
   150→  const [internalOpen, setInternalOpen] = useState(defaultOpen);
   151→  const isOpen = open ?? internalOpen;

   157→    <Card id={id} className="rounded-xl">
   158→      <Collapsible open={isOpen} onOpenChange={setOpen}>
   159→        <div className="flex items-center justify-between gap-3 border-b px-4 py-3">
   160→          <div className="min-w-0">

   162→              <p className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">

   167→              {title}
   168→            </h3>
   169→            {description ? (
   170→              <p className="mt-0.5 hidden text-xs text-muted-foreground sm:block">

   316→function ColumnsTable({ table }: { table: IntakeTableDto }) {
   317→  if (table.columns.length === 0) {
   318→    return (
   319→      <p className="text-xs text-muted-foreground">No column metadata.</p>
   320→    );
   321→  }
   322→  return (
   323→    <div className="wedjat-scroll overflow-x-auto">
   324→      <Table className="min-w-[720px]">
   325→        <TableHeader>
   326→          <TableRow>
   327→            <TableHead>Column</TableHead>
   328→            <TableHead>Type</TableHead>
   329→            <TableHead>Purpose</TableHead>
   330→            <TableHead>Flags</TableHead>
   331→            <TableHead>Enum values</TableHead>
   332→            <TableHead>Null %</TableHead>
   333→          </TableRow>
   334→        </TableHeader>
   335→        <TableBody>
   336→          {table.columns.map((col) => {
   337→            const nullPct =
   338→              col.nullPct === undefined
   339→                ? null
   340→                : Math.round((toRatio(col.nullPct) ?? 0) * 100);
   341→            return (
   342→              <TableRow key={col.name}>
   343→                <TableCell className="font-mono text-xs">{col.name}</TableCell>
   344→                <TableCell>
   345→                  <span className="font-mono text-[10px] text-muted-foreground">
   346→                    {col.rawType}
   347→                  </span>
   348→                  {col.normalizedType !== col.rawType ? (
   349→                    <span className="ml-1.5 font-mono text-[10px] text-primary">
   350→                      → {col.normalizedType}
   351→                    </span>
   352→                  ) : null}
   353→                </TableCell>
   354→                <TableCell className="max-w-56">
   355→                  <p className="truncate text-xs text-muted-foreground" title={col.purpose}>
   356→                    {col.purpose}
   357→                  </p>
   358→                </TableCell>
   359→                <TableCell>
   360→                  <span className="flex items-center gap-1">
   361→                    {col.isPrimaryKey ? (
   362→                      <Badge
   363→                        variant="outline"
   364→                        className="border-emerald-500/30 bg-emerald-500/10 px-1 text-[9px] text-emerald-700 dark:text-emerald-400"
   365→                      >
   366→                        PK
   367→                      </Badge>
   368→                    ) : null}
   369→                    {col.isForeignKey ? (
   370→                      <Badge
   371→                        variant="outline"
   372→                        className="border-teal-500/30 bg-teal-500/10 px-1 text-[9px] text-teal-700 dark:text-teal-400"
   373→                      >
   374→                        FK
   375→                      </Badge>
   376→                    ) : null}
   377→                    {col.nullable ? (
   378→                      <Badge variant="outline" className="px-1 text-[9px] text-muted-foreground">
   379→                        NULL
   380→                      </Badge>
   381→                    ) : null}
   382→                  </span>
   383→                </TableCell>
   384→                <TableCell className="max-w-64">
   385→                  {col.enumValues && col.enumValues.length > 0 ? (
   386→                    <span className="flex flex-wrap items-center gap-1">
   387→                      {col.enumValues.slice(0, 6).map((v) => (
   388→                        <Chip key={v} mono>
   389→                          {v}
   390→                        </Chip>
   391→                      ))}
   392→                      {col.enumValues.length > 6 ? (
   393→                        <span className="text-[10px] text-muted-foreground">
   394→                          +{col.enumValues.length - 6}
   395→                        </span>
   396→                      ) : null}
   397→                    </span>
   398→                  ) : (
   399→                    <span className="text-xs text-muted-foreground/70">—</span>
   400→                  )}
   401→                </TableCell>
   402→                <TableCell className="font-mono text-xs tabular-nums">
   403→                  {nullPct === null ? "—" : `${nullPct}%`}
   404→                </TableCell>
   405→              </TableRow>
   406→            );
   407→          })}
   408→        </TableBody>
   409→      </Table>
   410→    </div>
   411→  );
   412→}

   414→function SnapshotBrowser({ snapshot }: { snapshot: IntakeSnapshotDto }) {
   415→  const [expanded, setExpanded] = useState<Set<string>>(new Set());
   416→  const visible = snapshot.tables.slice(0, MAX_SNAPSHOT_TABLES);
   417→  const truncated =
   418→    snapshot.tablesTruncated === true || snapshot.tables.length > visible.length;
   419→
   420→  const toggle = (name: string) => {
   421→    setExpanded((cur) => {
   422→      const next = new Set(cur);
   423→      if (next.has(name)) next.delete(name);
   424→      else next.add(name);
   425→      return next;
   426→    });
   427→  };
   428→
   429→  return (
   430→    <div className="space-y-3">
   431→      <div className="grid gap-2 text-xs text-muted-foreground sm:grid-cols-3">
   432→        <p>
   433→          engine: <span className="font-mono text-foreground">{snapshot.engine}</span>
   434→        </p>
   435→        <p>
   436→          detection:{" "}
   437→          <span className="font-mono text-foreground">
   438→            {snapshot.detection.method} (
   439→            {Math.round((toRatio(snapshot.detection.confidence) ?? 0) * 100)}%)
   440→          </span>
   441→        </p>
   442→        <p>
   443→          snapshot v<span className="font-mono text-foreground">{snapshot.version}</span>{" "}
   444→          · {formatCount(snapshot.tablesCount)} tables ·{" "}
   445→          {formatCount(snapshot.columnsCount)} columns
   446→        </p>
   447→      </div>
   448→      {snapshot.detection.detail ? (
   449→        <p className="rounded-lg bg-muted/50 p-2.5 text-xs text-muted-foreground">
   450→          {snapshot.detection.detail}
   451→        </p>
   452→      ) : null}
   453→
   454→      <div className="wedjat-scroll overflow-x-auto">
   455→        <Table className="min-w-[640px]">
   456→          <TableHeader>
   457→            <TableRow>
   458→              <TableHead className="w-8" aria-label="Expand" />
   459→              <TableHead>Table</TableHead>
   460→              <TableHead>Rows</TableHead>
   461→              <TableHead>Purpose</TableHead>
   462→              <TableHead>Canonical entity</TableHead>
   463→              <TableHead>Primary key</TableHead>
   464→            </TableRow>
   465→          </TableHeader>
   466→          <TableBody>
   467→            {visible.map((t) => {
   468→              const isOpen = expanded.has(t.name);
   469→              return (
   470→                <Fragment key={t.name}>
   471→                  <TableRow
   472→                    className={cn(
   473→                      "cursor-pointer",
   474→                      isOpen && "border-b-0 bg-muted/30 hover:bg-muted/30",
   475→                    )}
   476→                    onClick={() => toggle(t.name)}
   477→                  >
   478→                    <TableCell>
   479→                      <button
   480→                        type="button"
   481→                        aria-expanded={isOpen}
   482→                        aria-label={`${isOpen ? "Collapse" : "Expand"} columns of ${t.name}`}
   483→                        className="flex size-6 items-center justify-center rounded"
   484→                        onClick={(e) => {
   485→                          e.stopPropagation();
   486→                          toggle(t.name);
   487→                        }}
   488→                      >
   489→                        <ChevronRight
   490→                          aria-hidden="true"
   491→                          className={cn(
   492→                            "size-3.5 text-muted-foreground transition-transform",
   493→                            isOpen && "rotate-90",
   494→                          )}
   495→                        />
   496→                      </button>
   497→                    </TableCell>
   498→                    <TableCell className="font-mono text-xs font-medium">
   499→                      {t.name}
   500→                    </TableCell>
   501→                    <TableCell className="font-mono text-xs tabular-nums">
   502→                      {formatCount(t.rowCount)}
   503→                    </TableCell>
   504→                    <TableCell className="max-w-64">
   505→                      <p className="truncate text-xs text-muted-foreground" title={t.purpose}>
   506→                        {t.purpose}
   507→                      </p>
   508→                    </TableCell>
   509→                    <TableCell>
   510→                      {t.canonicalEntity ? (
   511→                        <Badge
   512→                          variant="outline"
   513→                          className="border-primary/30 bg-primary/10 text-[10px] text-primary"
   514→                        >
   515→                          {t.canonicalEntity}
   516→                        </Badge>
   517→                      ) : (
   518→                        <span className="text-xs text-muted-foreground/70">—</span>
   519→                      )}
   520→                    </TableCell>
   521→                    <TableCell>
   522→                      <Chip mono>{t.primaryKey.join(", ") || "—"}</Chip>
   523→                    </TableCell>
   524→                  </TableRow>
   525→                  {isOpen ? (
   526→                    <TableRow className="hover:bg-transparent">
   527→                      <TableCell colSpan={6} className="bg-muted/30 p-4">
   528→                        <ColumnsTable table={t} />
   529→                      </TableCell>
   530→                    </TableRow>
   531→                  ) : null}
   532→                </Fragment>
   533→              );
   534→            })}
   535→          </TableBody>
   536→        </Table>
   537→      </div>
   538→      {truncated ? (
   539→        <p className="text-xs text-muted-foreground">
   540→          Showing the first {MAX_SNAPSHOT_TABLES} of{" "}
   541→          {formatCount(snapshot.tablesCount)} tables — the snapshot is truncated
   542→          for display; full metadata is preserved in the store.
   543→        </p>
   544→      ) : null}
   545→    </div>
   546→  );
   547→}
